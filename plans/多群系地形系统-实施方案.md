## 阶段 S1：噪声框架

### 目标

搭建 5 层噪声管线，生成有大陆/海洋/山脉自然起伏的地形。

### Step 1.1：创建噪声管理器

**新建文件**：`src/terrain_noise.rs`

```rust
//! 多层噪声地形生成系统
//!
//! 5 个独立噪声层驱动地形生成：
//! - continentalness：大陆性（海洋 vs 陆地）
//! - erosion：侵蚀程度
//! - ridge：山脊线
//! - temperature：温度
//! - vegetation：植被/湿度

use noise::{Fbm, Ridged, Simplex, NoiseFn};
use std::sync::OnceLock;

pub const SEA_LEVEL: i32 = 63;
pub const OCEAN_THRESHOLD: f64 = -0.3;
pub const DEEP_OCEAN_THRESHOLD: f64 = -0.5;
pub const CONTINENT_THRESHOLD: f64 = 0.2;

pub const TERRAIN_SEED: u32 = 12345;

/// 5 层噪声管理器
pub struct TerrainNoise {
    pub continentalness: Fbm<Simplex>,
    pub erosion: Fbm<Simplex>,
    pub ridge: Ridged<Simplex>,
    pub temperature: Fbm<Simplex>,
    pub vegetation: Fbm<Simplex>,
}

impl TerrainNoise {
    pub fn new(seed: u32) -> Self {
        Self {
            continentalness: Fbm::<Simplex>::new(seed)
                .set_octaves(6)
                .set_frequency(0.0008)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
            erosion: Fbm::<Simplex>::new(seed.wrapping_add(1))
                .set_octaves(6)
                .set_frequency(0.0015)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
            ridge: Ridged::<Simplex>::new(seed.wrapping_add(2))
                .set_octaves(4)
                .set_frequency(0.001)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
            temperature: Fbm::<Simplex>::new(seed.wrapping_add(3))
                .set_octaves(4)
                .set_frequency(0.0006)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
            vegetation: Fbm::<Simplex>::new(seed.wrapping_add(4))
                .set_octaves(4)
                .set_frequency(0.0006)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
        }
    }

    /// 采样所有噪声层
    pub fn sample_all(&self, wx: f64, wz: f64) -> NoiseSample {
        NoiseSample {
            continentalness: self.continentalness.get([wx, wz]),
            erosion: self.erosion.get([wx, wz]),
            ridge: self.ridge.get([wx, wz]),
            temperature: self.temperature.get([wx, wz]),
            vegetation: self.vegetation.get([wx, wz]),
        }
    }
}

/// 单次采样的所有噪声值
pub struct NoiseSample {
    pub continentalness: f64,
    pub erosion: f64,
    pub ridge: f64,
    pub temperature: f64,
    pub vegetation: f64,
}

/// 全局噪声缓存
static TERRAIN_NOISE: OnceLock<TerrainNoise> = OnceLock::new();

pub fn get_terrain_noise() -> &'static TerrainNoise {
    TERRAIN_NOISE.get_or_init(|| TerrainNoise::new(TERRAIN_SEED))
}
```

### Step 1.2：实现基础高度计算

**修改文件**：`src/terrain_noise.rs`（追加）

```rust
/// 地形类型（由大陆性决定）
#[derive(Debug, Clone, Copy, PartialEq)]
pub enum TerrainType {
    DeepOcean,
    Ocean,
    Coast,
    Inland,
}

impl NoiseSample {
    pub fn terrain_type(&self) -> TerrainType {
        if self.continentalness < DEEP_OCEAN_THRESHOLD {
            TerrainType::DeepOcean
        } else if self.continentalness < OCEAN_THRESHOLD {
            TerrainType::Ocean
        } else if self.continentalness < CONTINENT_THRESHOLD {
            TerrainType::Coast
        } else {
            TerrainType::Inland
        }
    }

    /// 计算基础地形高度
    pub fn compute_base_height(&self) -> f64 {
        let c = self.continentalness;
        let e = self.erosion;
        let r = self.ridge;

        // 海洋区域
        if c < OCEAN_THRESHOLD {
            let t = (c - DEEP_OCEAN_THRESHOLD) / (OCEAN_THRESHOLD - DEEP_OCEAN_THRESHOLD);
            return SEA_LEVEL as f64 - 80.0 * (1.0 - t); // -80 ~ 0
        }

        // 陆地基础高度
        let land_base = SEA_LEVEL as f64 + 20.0;

        // 山脊贡献
        let ridge_height = if r > 0.0 {
            r.powf(1.5) * 300.0
        } else {
            r * 50.0
        };

        // 侵蚀调制
        let erosion_factor = 1.0 - (e.max(0.0) * 0.6);

        land_base + ridge_height * erosion_factor
    }
}
```

```rust
//! 生物群系判定系统
//!
//! 三轴映射：温度 × 植被 × 海拔 → BiomeType

use crate::terrain_noise::{NoiseSample, SEA_LEVEL, TerrainType};

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum BiomeType {
    DeepOcean,
    Ocean,
    Plains,
    Desert,
    Forest,
    Mountains,
    SnowPlains,
}

/// 温度分级（5 级）
fn temperature_index(t: f64) -> u8 {
    if t < -0.48 { 1 }
    else if t < -0.18 { 2 }
    else if t < 0.17 { 3 }
    else if t < 0.52 { 4 }
    else { 5 }
}

/// 植被分级（5 级）
fn vegetation_index(v: f64) -> u8 {
    if v < -0.38 { 1 }
    else if v < -0.13 { 2 }
    else if v < 0.07 { 3 }
    else if v < 0.27 { 4 }
    else { 5 }
}

/// 海拔分级（4 级）
fn elevation_index(height: i32) -> u8 {
    if height < SEA_LEVEL { 1 }
    else if height < 100 { 2 }
    else if height < 180 { 3 }
    else { 4 }
}

/// 根据噪声采样和地表高度判定生物群系
pub fn get_biome(sample: &NoiseSample, surface_height: i32) -> BiomeType {
    // 海洋覆盖
    if sample.continentalness < -0.5 {
        return BiomeType::DeepOcean;
    }
    if sample.continentalness < -0.3 {
        return BiomeType::Ocean;
    }

    // 山脉覆盖（海拔 > 180）
    if surface_height > 180 {
        return BiomeType::Mountains;
    }

    // 温度×植被矩阵
    let ti = temperature_index(sample.temperature);
    let vi = vegetation_index(sample.vegetation);

    match (ti, vi) {
        // 极寒
        (1, _) => BiomeType::SnowPlains,
        // 寒冷
        (2, 1..=2) => BiomeType::SnowPlains,
        (2, _) => BiomeType::Forest,
        // 温和
        (3, 1) => BiomeType::Plains,
        (3, 2) => BiomeType::Plains,
        (3, _) => BiomeType::Forest,
        // 温暖
        (4, 1) => BiomeType::Desert,
        (4, 2) => BiomeType::Desert,
        (4, 3) => BiomeType::Plains,
        (4, _) => BiomeType::Forest,
        // 炎热
        (5, 1..=3) => BiomeType::Desert,
        (5, 4) => BiomeType::Plains,
        (5, _) => BiomeType::Forest,
        _ => BiomeType::Plains,
    }
}

impl BiomeType {
    /// 地表方块
    pub fn surface_block(self, world_y: i32) -> u8 {
        match self {
            BiomeType::DeepOcean | BiomeType::Ocean => {
                if world_y == SEA_LEVEL - 1 { 4 } else { 2 } // 沙/石
            }
            BiomeType::Plains | BiomeType::Forest => 1, // 草
            BiomeType::Desert => 4, // 沙
            BiomeType::Mountains => {
                if world_y > 200 { 11 } else { 2 } // 岩石/石
            }
            BiomeType::SnowPlains => 9, // 雪草
        }
    }

    /// 地下层方块
    pub fn subsurface_block(self, depth: i32) -> u8 {
        match self {
            BiomeType::Desert => {
                if depth < 4 { 4 } else { 8 } // 沙/沙石
            }
            BiomeType::Mountains => {
                if depth < 2 { 10 } else { 2 } // 砂砾/石
            }
            _ => {
                if depth < 4 { 3 } else { 2 } // 泥土/石
            }
        }
    }

    /// 树木密度乘数
    pub fn tree_density_multiplier(self) -> f64 {
        match self {
            BiomeType::Forest => 2.0,
            BiomeType::Plains => 0.5,
            BiomeType::SnowPlains => 0.3,
            BiomeType::Desert => 0.0,
            BiomeType::Mountains => 0.1,
            _ => 0.0,
        }
    }
}
```

### Step 3.1：添加河流噪声

```rust
use noise::Ridged;

/// 河流噪声
static RIVER_NOISE: OnceLock<Ridged<Simplex>> = OnceLock::new();

pub fn get_river_noise() -> &'static Ridged<Simplex> {
    RIVER_NOISE.get_or_init(|| {
        Ridged::<Simplex>::new(TERRAIN_SEED.wrapping_add(100))
            .set_octaves(3)
            .set_frequency(0.0008)
            .set_lacunarity(2.0)
            .set_persistence(0.5)
    })
}

/// 判断是否在河道中，返回河道深度
pub fn river_depth_at(wx: f64, wz: f64, continentalness: f64) -> Option<i32> {
    if continentalness < OCEAN_THRESHOLD {
        return None; // 海洋中无河流
    }

    let river_val = get_river_noise().get([wx, wz]).abs();
    let river_width = 0.02;

    if river_val < river_width {
        let depth_factor = 1.0 - (river_val / river_width);
        let depth = (depth_factor * 8.0) as i32 + 2;
        Some(depth)
    } else {
        None
    }
}
```

```rust
pub fn fill_terrain(chunk: &mut Chunk, coord: &ChunkCoord) {
    let noise = get_terrain_noise();

    for z in 0..CHUNK_SIZE {
        for x in 0..CHUNK_SIZE {

            let surface_height = sample.compute_base_height() as i32;

            // 河流检测
            let river = river_depth_at(world_x, world_z, sample.continentalness);

            for y in 0..CHUNK_SIZE {
                let world_y = coord.cy as i32 * CHUNK_SIZE as i32 + y as i32;

                // 河道处理
                if let Some(river_depth) = river {
                    let river_bottom = surface_height - river_depth;

                    if world_y > surface_height && world_y <= SEA_LEVEL {
                        // 水面
                        chunk.set(x, y, z, 5);
                        continue;
                    } else if world_y <= surface_height && world_y > river_bottom {
                        // 河道挖掘
                        chunk.set(x, y, z, 0); // 空气
                        continue;
                    } else if world_y == river_bottom {
                        // 河床
                        chunk.set(x, y, z, 10); // 砂砾
                        continue;
                    }
                }

                // 普通地形
                if world_y > surface_height {
                    if world_y <= SEA_LEVEL && surface_height < SEA_LEVEL {
                        chunk.set(x, y, z, 5); // 海洋水
                    }
                    continue;
                }

                let depth = surface_height - world_y;
                let block_id = if depth == 0 {
                    biome.surface_block(world_y)
                } else {
                    biome.subsurface_block(depth)
                };

                chunk.set(x, y, z, block_id);
            }
        }
    }
}
```

```rust
/// 洞穴噪声配置
pub struct CaveNoise {
    pub cheese: Fbm<Simplex>,
    pub spaghetti: Fbm<Simplex>,
    pub noodle: Fbm<Simplex>,
}

impl CaveNoise {
    pub fn new(seed: u32) -> Self {
        Self {
            cheese: Fbm::<Simplex>::new(seed.wrapping_add(200))
                .set_octaves(3)
                .set_frequency(0.02)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
            spaghetti: Fbm::<Simplex>::new(seed.wrapping_add(201))
                .set_octaves(4)
                .set_frequency(0.03)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
            noodle: Fbm::<Simplex>::new(seed.wrapping_add(202))
                .set_octaves(2)
                .set_frequency(0.05)
                .set_lacunarity(2.0)
                .set_persistence(0.5),
        }
    }

    /// 计算洞穴密度（>0 表示应该挖空）
    pub fn density(&self, wx: f64, wy: f64, wz: f64, depth_from_surface: i32) -> f64 {
        // 地表附近不生成洞穴
        if depth_from_surface < 10 {
            return 0.0;
        }

        let depth_factor = ((depth_from_surface - 10) as f64 / 50.0).clamp(0.0, 1.0);

        // Cheese：大型空腔
        let cheese_val = self.cheese.get([wx * 0.7, wy as f64 * 0.8, wz * 0.7]);
        let cheese = if cheese_val > 0.6 - depth_factor * 0.3 {
            cheese_val
        } else {
            0.0
        };

        // Spaghetti：蜿蜒隧道
        let spaghetti_val = self.spaghetti.get([wx, wy as f64 * 1.2, wz]);
        let spaghetti = if spaghetti_val.abs() < 0.15 {
            0.5 - spaghetti_val.abs() * 3.0
        } else {
            0.0
        };

        // Noodle：细小裂隙
        let noodle_val = self.noodle.get([wx * 1.5, wy as f64 * 2.0, wz * 1.5]);
        let noodle = if noodle_val.abs() < 0.08 {
            0.3 - noodle_val.abs() * 3.0
        } else {
            0.0
        };

        (cheese.max(spaghetti).max(noodle)) * depth_factor
    }
}
```

```rust
                    biome.subsurface_block(depth)
                };

                chunk.set(x, y, z, block_id);
            }
        }
    }
}
```
